void initializeRainMehedi() {
    for (int i = 0; i < NUM_RAIN_MEHEDI; i++) {
        RainMehediX[i] = rand() % 1280;
        RainMehediY[i] = rand() % 720;
    }
    rainInitialized = true;
}
void RainMehedi() {
    if (!isRaining) return;
    if (!rainInitialized) initializeRainMehedi();

    glColor3f(0.7f, 0.7f, 0.9f);  // Soft bluish lines

    glBegin(GL_LINES);
    for (int i = 0; i < NUM_RAIN_MEHEDI; i++) {
        glVertex2f(RainMehediX[i], RainMehediY[i]);
        glVertex2f(RainMehediX[i] - 4.0f, RainMehediY[i] - 10.0f);
    }
    glEnd();
}
void updateRainMehedi(int value) {
    if (isRaining) {
        for (int i = 0; i < NUM_RAIN_MEHEDI; i++) {
            RainMehediX[i] -= 4.0f;
            RainMehediY[i] -= 10.0f;

            if (RainMehediY[i] < 0 || RainMehediX[i] < 0) {
                RainMehediX[i] = rand() % 1280;
                RainMehediY[i] = 720 + rand() % 100;
            }
        }
        glutPostRedisplay();
    }

    glutTimerFunc(16, updateRainMehedi, 0);
}

void updateBird(int value) {

    if (birdActive) {
        birdOffsetX += 1.50f;

        // Reset bird when it flies off-screen
        if (birdOffsetX > 1000) {
            birdOffsetX = -300.0f;

            // If night has started, don't launch a new bird
            if (transitioningToNight) {
                birdActive = false; // Deactivate bird until day returns
            }
        }

        // Wing flap animation
        if (wingUp) {
            wingFlap += 0.7f;
            if (wingFlap > 15) wingUp = false;
        } else {
            wingFlap -= 0.7f;
            if (wingFlap < -15) wingUp = true;
        }
    } else {
        // If it's day again, allow new birds
        if (!transitioningToNight) {
            birdActive = true;
            birdOffsetX = -300.0f; // left theke restart nibe
        }
    }

    glutPostRedisplay();
    glutTimerFunc(16, updateBird, 0);
}

void drawSun(float cx, float cy, float radius, float alpha = 1.0f, int segments = 100) {

float r = 1.0f;
float g = 0.3f + 0.5f * sunAlpha;
float b = 0.0f + 0.1f * sunAlpha;
glColor4f(r, g, b, alpha);

    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    glBegin(GL_TRIANGLE_FAN);
    glVertex2f(cx, cy);
    for (int i = 0; i <= segments; ++i) {
        float angle = 2.0f * M_PI * i / segments;
        float x = cx + cos(angle) * radius;
        float y = cy + sin(angle) * radius;
        glVertex2f(x, y);
    }
    glEnd();

    glDisable(GL_BLEND);
}

void updateSun(int value) {
    if (!sunVisible) {
        glutTimerFunc(33, updateSun, 0);
        return;
    }

    float speed = 0.5f;
    float dx = targetSunX - sunX;
    float dy = targetSunY - sunY;

    if (sunShouldSet && fabs(dx) < 1.0f && fabs(dy) < 1.0f) {
        sunAlpha -= 0.01f;
        if (sunAlpha <= 0.0f) {
            sunAlpha = 0.0f;
            sunVisible = false;

            glutTimerFunc(20000, [](int) {
    transitioningToNight = true;

    glutTimerFunc(10000, [](int) {
        moonVisible = true;
        moonAlpha = 0.0f;
        updateMoon(0);
        glutPostRedisplay();
    }, 0);
}, 0);
            }
        }
     else if (sunShouldSet) {
        float distance = sqrt(dx * dx + dy * dy);
        if (distance > 0.0f) {
            sunX += speed * dx / distance;
            sunY += speed * dy / distance;
        }
    }

    glutPostRedisplay();
    glutTimerFunc(33, updateSun, 0);
}

GLuint loadBMP_custom(const char* imagepath) {
    FILE* file = fopen(imagepath, "rb");
    if (!file) {
        std::cout << "Image not found!" << std::endl;
        return 0;
    }

    unsigned char header[54];
    fread(header, 1, 54, file); // BMP header

    unsigned int dataPos = *(int*)&(header[0x0A]);
    unsigned int width = *(int*)&(header[0x12]);
    unsigned int height = *(int*)&(header[0x16]);
    unsigned int imageSize = *(int*)&(header[0x22]);

    if (imageSize == 0) imageSize = width * height * 3;
    if (dataPos == 0) dataPos = 54;

    unsigned char* data = new unsigned char[imageSize];
    fread(data, 1, imageSize, file);
    fclose(file);

    GLuint textureID;
    glGenTextures(1, &textureID);
    glBindTexture(GL_TEXTURE_2D, textureID);

    // Create texture
    glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB, width, height,
                 0, GL_BGR, GL_UNSIGNED_BYTE, data);

    // Texture options
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);

    delete[] data;
    return textureID;

}

            float rad = i * 3.14159f / 180.0f;
            glVertex2f(cx + cos(rad) * r, cy + sin(rad) * r);
        }
    glEnd();
}
const int NUM_STARS = 50;
float starX[NUM_STARS], starY[NUM_STARS];
float starBrightness[NUM_STARS];
bool starsInitialized = false;

void drawStars() {
    if (!isNightA) return;

    if (!starsInitialized) {
        for (int i = 0; i < NUM_STARS; i++) {
            starX[i] = rand() % 200;
            starY[i] = 150 + rand() % 50;
            starBrightness[i] = 0.5f + ((rand() % 50) / 100.0f);
        }
        starsInitialized = true;
    }

    for (int i = 0; i < NUM_STARS; i++) {
        // Flicker brightness
        starBrightness[i] += ((rand() % 3) - 1) * 0.05f; // small change up or down
        if (starBrightness[i] < 0.3f) starBrightness[i] = 0.3f;
        if (starBrightness[i] > 1.0f) starBrightness[i] = 1.0f;

        glColor3f(starBrightness[i], starBrightness[i], starBrightness[i]);
        glBegin(GL_POINTS);
            glVertex2f(starX[i], starY[i]);
        glEnd();
    }
}

void drawCircle(float cx, float cy, float r, int num_segments) {
    glBegin(GL_POLYGON);
    for (int i = 0; i < num_segments; ++i) {
        float theta = 2.0f * 3.1415926f * float(i) / float(num_segments);
        float x = r * cosf(theta);
        float y = r * sinf(theta);
        glVertex2f(x + cx, y + cy);
    }
    glEnd();
}

void updateSkyColorBasedOnSun() {
    float sunStartY = 175.0f;  // Highest sun position (day)
    float sunEndY = 0.0f;      // Lowest sun position (night)

    float t;
    if (SunAYA > sunStartY) {
        t = 0.0f;
    } else if (SunAYA < sunEndY) {
        t = 1.0f;
    } else {
        t = (sunStartY - SunAYA) / (sunStartY - sunEndY);
    }

    // Day color (light blue)
    float daySky[3] = {0.5f, 0.8f, 1.0f};
    // Night color (dark blue)
    float nightSky[3] = {0.05f, 0.05f, 0.1f};

    float skyR = daySky[0] * (1.0f - t) + nightSky[0] * t;
    float skyG = daySky[1] * (1.0f - t) + nightSky[1] * t;
    float skyB = daySky[2] * (1.0f - t) + nightSky[2] * t;

    glClearColor(skyR, skyG, skyB, 1.0f);
}

void updateSunA(int value) {
    if (sunIsSetting) {
        SunAYA -= 0.7f;  // Faster descent to fully set below screen
        if (SunAYA < 0.0f) {  // Sun fully set below screen
            sunIsSetting = false;
            isNightA = true;
            moonIsRising = true;
            MoonAYA = 220.0f;  // Moon starts above screen
        }
    }
    else if (sunIsRising) {
        SunAYA -= 0.7f;  // Sun rising *downwards* into view (because Y decreases going down)
        if (SunAYA <= 175.0f) {  // Sun reached visible daytime position
            sunIsRising = false;
            isNightA = false;
            sunIsSetting = true;  // Prepare for next set
        }
    }
    glutPostRedisplay();
    glutTimerFunc(30, updateSunA, 0);
}

void updateMoonA(int value) {
    if (moonIsRising) {
        MoonAYA -= 0.7f;  // Moon moves down into visible range
        if (MoonAYA <= 0.0f) {  // Moon fully visible (you can tweak this)
            moonIsRising = false;
            moonIsSetting = true;
        }
    }
    else if (moonIsSetting) {
        MoonAYA -= 2.0f;  // Moon sets below screen
        if (MoonAYA < 0.0f) {  // Moon fully set below screen
            moonIsSetting = false;
            sunIsRising = true;  // Start sun rising from top
            SunAYA = 220.0f;    // Reset sun starting position above screen
            isNightA = false;
        }
    }
    glutPostRedisplay();
    glutTimerFunc(30, updateMoonA, 0);
}

void updateRainA(int value) {
    if (isRainAing) {
        for (int i = 0; i < NUM_RainADROPSA; i++) {
            RainAY[i] -= 2.5f;
            if (RainAY[i] < 0) {
                RainAY[i] = 200.0f;
                RainAX[i] = rand() % 200;
            }
        }
    }
    glutPostRedisplay();
    glutTimerFunc(30, updateRainA, 0);
}

void draw_circle(float x, float y, float radius) {
    glMatrixMode(GL_MODELVIEW);
    glPushMatrix();
    glLoadIdentity();
    glTranslatef(x, y, 0.0f);
    static const int circle_points = 100;
    static const float angle = 2.0f * 3.1416f / circle_points;

    glBegin(GL_POLYGON);
    double angle1=0.0;
    glVertex2d(radius * cos(0.0) , radius * sin(0.0));
    int i;
    for (i=0; i<circle_points; i++) {
        glVertex2d(radius * cos(angle1), radius *sin(angle1));
        angle1 += angle;
    }
    glEnd();
    glPopMatrix();
}

void drawWheel(float centerX, float centerY, float radius) {
    glBegin(GL_TRIANGLE_FAN);
        glVertex2f(centerX, centerY);

            glVertex2f(centerX + cos(angle) * radius, centerY + sin(angle) * radius);
        }
    glEnd();
}

void circle(GLfloat rx, GLfloat ry, GLfloat cx, GLfloat cy) {
    glBegin(GL_POLYGON);
    for (int i = 0; i <= 360; i++) {

        float x = rx * cos(angle);
        float y = ry * sin(angle);
        glVertex2f(x + cx, y + cy);
    }
    glEnd();
}

void idle() {
    // Move Boat 1: left to right
    boat1Shift += boatSpeed1;
    if (boat1Shift > 400)    // Off-screen right
        boat1Shift = -200;   // Reset to left

    // Move Boat 2: right to left
    boat2Shift += boatSpeed2;
    if (boat2Shift < -200)   // Off-screen left
        boat2Shift = 700;    // Reset to right

    // Windmill blade rotation (if using)
    bladeAngle += 1.0f;
    if (bladeAngle > 360)
        bladeAngle -= 360;

    shipShift += 0.15;
    if (shipShift > 700)
        shipShift = -350;

    cloudShift += 0.1f;
    if (cloudShift > 600)
        cloudShift = -300;

    glutPostRedisplay();
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();

    // Set correct projection for each scene
    switch (currentScene) {
        case 0: // Landscape Scene
            gluOrtho2D(0.0, 1000.0, 0.0, 600.0);
            break;
        case 2: // Airport Scene
            gluOrtho2D(0.0, 200.0, 0.0, 200.0);
            break;
        case 1:
            gluOrtho2D(0.0, 800.0, 0.0, 800.0);
            break;
        case 3:
            gluOrtho2D(0, 500, 0, 500);
            break;
    }

    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();

    // Draw the current scene
    switch (currentScene) {
        case 0:
            drawLandscapeScene();
            break;
        case 2:
            drawAirportScene();
            break;
        case 1:
            drawHighwaySceneUnified();
            break;
        case 3:
            drawDipu();
            break;
    }

    glFlush();
}
